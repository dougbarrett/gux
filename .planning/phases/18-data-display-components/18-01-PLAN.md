   func getInitials(name string) string {
       if name == "" {
           return "?"
       }
       parts := strings.Fields(name)
       if len(parts) == 0 {
           return "?"
       }
       if len(parts) == 1 {
           if len(parts[0]) > 0 {
               return strings.ToUpper(string(parts[0][0]))
           }
           return "?"
       }
       first := string(parts[0][0])
       last := string(parts[len(parts)-1][0])
       return strings.ToUpper(first + last)
   }
