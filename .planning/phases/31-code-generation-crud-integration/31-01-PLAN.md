4. **Add file cleanup helper method on App** (before handleDelete):

```go
// deleteFileIfExists deletes a file from storage if the key is non-empty.
// Honors BeforeFileDelete hook. Logs warnings on failure but does not return errors
// (file cleanup should not prevent record operations from succeeding).
func (a *App) deleteFileIfExists(model CRUDModel, key string) {
    if key == "" || a.storage == nil {
        return
    }
    if model.OnBeforeFileDelete != nil {
        if err := model.OnBeforeFileDelete(key); err != nil {
            return // Hook prevented deletion
        }
    }
    if !model.DisableAutoCleanup {
        ctx := context.Background()
        if err := a.storage.Delete(ctx, key); err != nil {
            // Log but don't fail -- file cleanup is best-effort
            fmt.Printf("gux: warning: failed to delete file %s: %v\n", key, err)
        }
    }
}

// getFileFieldValues reads the current values of file fields from a model instance using reflection.
func getFileFieldValues(item interface{}, fileFields []string) map[string]string {
    values := make(map[string]string)
    v := reflect.ValueOf(item)
    if v.Kind() == reflect.Ptr {
        v = v.Elem()
    }
    for _, fieldName := range fileFields {
        f := v.FieldByName(fieldName)
        if f.IsValid() && f.Kind() == reflect.String {
            values[fieldName] = f.String()
        }
    }
    return values
}
```
